synthesize.js
JavaScript
const text = "Experience the next-gen speech synthesis.";
const voiceId = "nova-female-warm";

// Synthesize script using VoiceNova API
const response = await fetch('https://api.voicenova.ai/v1/tts', {
    method: 'POST',
    headers: {
        'Authorization': 'Bearer VN_API_KEY',
        'Content-Type': 'application/json'
    },
    body: JSON.stringify({ text, voiceId, format: 'mp3' })
});

const audioBlob = await response.blob();
const audioUrl = URL.createObjectURL(audioBlob);
new Audio(audioUrl).play();
